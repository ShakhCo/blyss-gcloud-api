# Bulk SMS Campaigns Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship a per-creator bulk SMS feature: AI polishes example messages into Eskiz-moderable templates, operator approves in Firestore console, creators send to up to 100 recipients pulled from that business's bookings, every send is logged.

**Architecture:** Backend adds one Express router mounted under `/businesses/:businessId/sms/*` (so multi-business owners pick which business they're acting on). A small `smsContext` resolver authorizes the caller as either the business owner or an employee at that business, returning `{ creator_id, creator_type, business_id }`. Templates and campaigns live in two new top-level Firestore collections (`sms_templates`, `sms_campaigns`). AI polishing is a single OpenAI call with server-side rule re-validation. Frontend adds one new authenticated route in `blyss-business-app` with three pages (templates / send / history).

**Tech Stack:** Express 5, Zod, Firestore, OpenAI (`gpt-4o-mini`), Eskiz SMS, Vitest + supertest, React Router v7, antd-mobile, `@tanstack/react-query`.

**Spec:** [`docs/superpowers/specs/2026-05-23-bulk-sms-design.md`](../specs/2026-05-23-bulk-sms-design.md)

```javascript
import OpenAI from 'openai';

let client = null;
function getClient() {
    if (!process.env.OPENAI_API_KEY) {
        const err = new Error('OpenAI not configured');
        err.code = 'AI_UNAVAILABLE';
        throw err;
    }
    if (!client) client = new OpenAI();
    return client;
}

const SYSTEM_PROMPT = `You polish SMS marketing messages for Uzbek barbershops and salons sending via the Eskiz SMS gateway. Output ONE message body, plain text, no more than 160 characters total, no emojis, no URLs, no phone numbers other than the business's own, no all-caps shouting. Preserve the user's language (Uzbek/Russian/mixed) and intent. Append "BLYSS" on its own line at the end if not already present. Return only the polished message, no preamble.`;

const URL_REGEX = /\b(https?:\/\/|www\.)\S+|\b[\w-]+\.(com|uz|ru|net|org|io|app|me)\b/i;

export function _validatePolishedOutput(text) {
    const trimmed = (text || '').trim();
    if (trimmed.length === 0) {
        const err = new Error('Polished message is empty');
        err.code = 'AI_OUTPUT_INVALID';
        err.rule = 'empty';
        throw err;
    }
    if (trimmed.length > 160) {
        const err = new Error(`Polished message exceeds 160 chars (${trimmed.length})`);
        err.code = 'AI_OUTPUT_INVALID';
        err.rule = 'too_long';
        throw err;
    }
    if (URL_REGEX.test(trimmed)) {
        const err = new Error('Polished message contains a URL');
        err.code = 'AI_OUTPUT_INVALID';
        err.rule = 'no_urls';
        throw err;
    }
}

export async function polishSmsText(exampleText) {
    const openai = getClient();
    const response = await openai.chat.completions.create({
        model: 'gpt-4o-mini',
        temperature: 0.3,
        max_tokens: 200,
        messages: [
            { role: 'system', content: SYSTEM_PROMPT },
            { role: 'user', content: exampleText },
        ],
    });
    const text = response.choices?.[0]?.message?.content ?? '';
    _validatePolishedOutput(text);
    return text.trim();
}
```

```javascript
import { db } from '../db/db.js';

export async function resolveSmsContext(user, businessId) {
    if (!user || !businessId) {
        const err = new Error('Missing user or businessId');
        err.code = 'FORBIDDEN';
        throw err;
    }

    const businessDoc = await db.collection('businesses').doc(businessId).get();
    if (!businessDoc.exists) {
        const err = new Error('Business not found');
        err.code = 'NOT_FOUND';
        throw err;
    }

    const businessData = businessDoc.data();

    if (user.user_type === 'business_owner' && businessData.business_owner_id === user.id) {
        return {
            creator_id: user.id,
            creator_type: 'business_owner',
            business_id: businessId,
        };
    }

    if (user.phone_number) {
        const empSnap = await db
            .collectionGroup('employees')
            .where('phone_number', '==', user.phone_number)
            .limit(20)
            .get();

        if (!empSnap.empty) {
            const match = empSnap.docs.find(
                (d) => d.ref.parent.parent && d.ref.parent.parent.id === businessId,
            );
            if (match) {
                return {
                    creator_id: match.id,
                    creator_type: 'employee',
                    business_id: businessId,
                };
            }
        }
    }

    const err = new Error('Not authorized for this business');
    err.code = 'FORBIDDEN';
    throw err;
}
```

```javascript
    const { business_id } = req.smsCtx;

    const snap = await db
        .collection('bookings')
        .where('business_id', '==', business_id)
        .get();

    const byPhone = new Map();
    for (const doc of snap.docs) {
        const b = doc.data();
        if (!b.phone_number) continue;
        const visitDate =
            b.booking_date?.toDate?.() ??
            b.created_at?.toDate?.() ??
            null;
        const existing = byPhone.get(b.phone_number);
        if (!existing) {
            byPhone.set(b.phone_number, {
                phone_number: b.phone_number,
                name: b.client_name || null,
                last_visit_at: visitDate,
                visit_count: 1,
            });
        } else {
            existing.visit_count += 1;
            if (visitDate && (!existing.last_visit_at || visitDate > existing.last_visit_at)) {
                existing.last_visit_at = visitDate;
                if (b.client_name) existing.name = b.client_name;
            }
        }
    }

    const items = Array.from(byPhone.values())
        .sort((a, b) => (b.last_visit_at?.getTime() ?? 0) - (a.last_visit_at?.getTime() ?? 0))
        .slice(0, 500)
        .map((r) => ({
            ...r,
            last_visit_at: r.last_visit_at?.toISOString() ?? null,
        }));
```

```javascript
async function sendWithConcurrency(phones, message, limit = 5) {
    const results = [];
    let cursor = 0;
    async function worker() {
        while (cursor < phones.length) {
            const idx = cursor++;
            const phone = phones[idx];
            try {
                const r = await sendSms(phone, message);
                results[idx] = { phone, success: !!r.success, error: r.error };
            } catch (err) {
                results[idx] = { phone, success: false, error: err.message };
            }
        }
    }
    await Promise.all(
        Array.from({ length: Math.min(limit, phones.length) }, () => worker()),
    );
    return results;
}

router.post('/send', validate(sendCampaignSchema), async (req, res) => {
    const { template_id, phone_numbers } = req.validated;
    const { creator_id, creator_type, business_id } = req.smsCtx;

    const tplRef = db.collection('sms_templates').doc(template_id);
    const tplSnap = await tplRef.get();
    if (!tplSnap.exists) {
        return res.status(404).json({ error: 'Template not found', error_code: 'NOT_FOUND' });
    }
    const tpl = tplSnap.data();
    if (tpl.business_id !== business_id) {
        return res.status(403).json({ error: 'Forbidden', error_code: 'FORBIDDEN' });
    }
    if (tpl.status !== 'confirmed') {
        return res
            .status(403)
            .json({ error: 'Template not approved', error_code: 'TEMPLATE_NOT_APPROVED' });
    }

    const results = await sendWithConcurrency(phone_numbers, tpl.polished_text, 5);
    const failures = results.filter((r) => !r.success);

    const campaignDoc = {
        business_id,
        sender_id: creator_id,
        sender_type: creator_type,
        template_id,
        message_snapshot: tpl.polished_text,
        recipient_count: phone_numbers.length,
        success_count: results.length - failures.length,
        failure_count: failures.length,
        errors: failures.slice(0, 20).map((f) => ({ phone: f.phone, error: f.error || 'unknown' })),
        sent_at: FieldValue.serverTimestamp(),
    };
    const ref = await db.collection('sms_campaigns').add(campaignDoc);

    const payload = {
        campaign_id: ref.id,
        sent: campaignDoc.success_count,
        failed: campaignDoc.failure_count,
        errors: campaignDoc.errors,
    };
```

```tsx
import { useState } from 'react';
import { Modal, TextArea, Button, Toast } from 'antd-mobile';

export function NewTemplateModal({
  visible,
  onClose,
}: {
  visible: boolean;
  onClose: () => void;
```

```tsx
import { useState, useMemo } from 'react';
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
import { Button, Checkbox, SearchBar, Toast } from 'antd-mobile';
import { smsApi, SmsTemplate, Recipient } from '../../lib/sms-api';
import { useActiveBusinessId } from '../../lib/active-business';

type Step = 'template' | 'recipients' | 'confirm';

export function SendWizard() {
  const businessId = useActiveBusinessId();
  const api = smsApi(businessId);
  const qc = useQueryClient();

  const [step, setStep] = useState<Step>('template');
  const [tplId, setTplId] = useState<string | null>(null);
  const [search, setSearch] = useState('');
  const [selected, setSelected] = useState<Set<string>>(new Set());

  const { data: templates = [] } = useQuery({
    queryKey: ['sms-templates', businessId, 'confirmed'],
    queryFn: () => api.listTemplates('confirmed'),
  });

  const { data: recipients = [] } = useQuery({
    queryKey: ['sms-recipients', businessId],
    queryFn: () => api.listRecipients(),
    enabled: step === 'recipients',
  });

  const filtered = useMemo(() => {
    const q = search.trim().toLowerCase();
    if (!q) return recipients;
    return recipients.filter(
      (r) =>
        r.phone_number.includes(q) ||
        (r.name && r.name.toLowerCase().includes(q)),
    );
  }, [recipients, search]);

  const tpl = templates.find((t) => t.id === tplId);

  const send = useMutation({
    mutationFn: () => api.send(tplId!, Array.from(selected)),
    onSuccess: (data) => {
      Toast.show({
        icon: data.failed > 0 ? 'fail' : 'success',
        content: `Sent ${data.sent} / Failed ${data.failed}`,
      });
      qc.invalidateQueries({ queryKey: ['sms-campaigns', businessId] });
      setStep('template');
      setTplId(null);
      setSelected(new Set());
    },
    onError: (e: any) => Toast.show({ icon: 'fail', content: e.message || 'Send failed' }),
  });

  function toggle(phone: string) {
    const next = new Set(selected);
    if (next.has(phone)) next.delete(phone);
    else if (next.size < 100) next.add(phone);
    else Toast.show({ content: '100 recipients max' });
    setSelected(next);
  }

  function selectAll() {
    const next = new Set<string>();
    for (const r of filtered.slice(0, 100)) next.add(r.phone_number);
    setSelected(next);
  }

  return (
    <div className="mt-3">
      {step === 'template' && (
        <div className="space-y-2">
          <h2 className="font-medium">Choose a confirmed template</h2>
          {templates.length === 0 && (
            <div className="text-gray-500 text-sm">
              No confirmed templates yet. Create one in the Templates tab.
            </div>
          )}
          {templates.map((t) => (
            <label key={t.id} className="border rounded p-3 block">
              <input
                type="radio"
                name="tpl"
                checked={tplId === t.id}
                onChange={() => setTplId(t.id)}
              />
              <span className="ml-2 whitespace-pre-wrap text-sm">{t.polished_text}</span>
            </label>
          ))}
          <Button
            color="primary"
            block
            disabled={!tplId}
            onClick={() => setStep('recipients')}
          >
            Next: pick recipients
          </Button>
        </div>
      )}

      {step === 'recipients' && (
        <div className="space-y-2">
          <SearchBar placeholder="Search name or phone" value={search} onChange={setSearch} />
          <div className="flex justify-between items-center">
            <div className="text-sm text-gray-600">{selected.size}/100 selected</div>
            <Button size="mini" onClick={selectAll}>
              Select all (cap 100)
            </Button>
          </div>
          <div className="max-h-80 overflow-y-auto border rounded">
            {filtered.map((r) => (
              <label
                key={r.phone_number}
                className="flex items-center gap-2 p-2 border-b last:border-b-0"
              >
                <Checkbox
                  checked={selected.has(r.phone_number)}
                  onChange={() => toggle(r.phone_number)}
                />
                <div className="flex-1">
                  <div className="text-sm">{r.name || '—'}</div>
                  <div className="text-xs text-gray-500">{r.phone_number}</div>
                </div>
                <div className="text-xs text-gray-400">×{r.visit_count}</div>
              </label>
            ))}
          </div>
          <div className="flex gap-2">
            <Button block onClick={() => setStep('template')}>
              Back
            </Button>
            <Button
              color="primary"
              block
              disabled={selected.size === 0}
              onClick={() => setStep('confirm')}
            >
              Next: confirm
            </Button>
          </div>
        </div>
      )}

      {step === 'confirm' && tpl && (
        <div className="space-y-3">
          <h2 className="font-medium">Confirm send</h2>
          <div className="border rounded p-3 bg-gray-50 whitespace-pre-wrap text-sm">
            {tpl.polished_text}
          </div>
          <div className="text-sm">
            Sending to <b>{selected.size}</b> recipients.
          </div>
          <div className="flex gap-2">
            <Button block onClick={() => setStep('recipients')} disabled={send.isPending}>
              Back
            </Button>
            <Button
              color="primary"
              block
              loading={send.isPending}
              onClick={() => send.mutate()}
            >
              Send now
            </Button>
          </div>
        </div>
      )}
    </div>
  );
}
```

```tsx
import { useState } from 'react';
import { useQuery } from '@tanstack/react-query';
import { smsApi, SmsCampaign } from '../../lib/sms-api';

function formatDate(iso: string | null) {
  if (!iso) return '—';
  return new Date(iso).toLocaleString();
}

export function CampaignsHistory() {
  const businessId = useActiveBusinessId();
  const { data, isLoading } = useQuery({
    queryKey: ['sms-campaigns', businessId],
    queryFn: () => smsApi(businessId).listCampaigns(),
  });

  if (isLoading) return <div className="mt-3">Loading…</div>;
  if (!data || data.length === 0)
    return <div className="mt-3 text-gray-500">No campaigns yet.</div>;

  return (
    <div className="mt-3 space-y-2">
      {data.map((c: SmsCampaign) => (
        <div key={c.id} className="border rounded p-3">
          <div className="flex justify-between text-sm">
            <div>{formatDate(c.sent_at)}</div>
            <div className="text-gray-600">
              {c.success_count}/{c.recipient_count} sent
              {c.failure_count > 0 && (
                <span className="text-red-600 ml-2">({c.failure_count} failed)</span>
              )}
            </div>
          </div>
          <div className="text-sm whitespace-pre-wrap mt-1">
            {expanded === c.id
              ? c.message_snapshot
              : c.message_snapshot.slice(0, 100) +
                (c.message_snapshot.length > 100 ? '…' : '')}
          </div>
          {c.message_snapshot.length > 100 && (
            <button
              type="button"
              className="text-xs text-blue-600 mt-1"
              onClick={() => setExpanded(expanded === c.id ? null : c.id)}
            >
              {expanded === c.id ? 'Show less' : 'Show more'}
            </button>
          )}
          {c.errors.length > 0 && expanded === c.id && (
            <div className="mt-2 text-xs">
              <div className="font-medium text-red-600">Errors:</div>
              <ul className="list-disc pl-5">
                {c.errors.map((e, i) => (
                  <li key={i}>
                    {e.phone}: {e.error}
                  </li>
                ))}
              </ul>
            </div>
          )}
        </div>
      ))}
    </div>
  );
}
```
